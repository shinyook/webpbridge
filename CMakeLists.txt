cmake_minimum_required(VERSION 3.20)
project(webpjni LANGUAGES C)
find_package(JNI REQUIRED)
# Ship the Windows CRT in the native binary rather than requiring a separate
# Visual C++ redistributable installation on consumer machines.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(WEBP_JNI_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(WEBP_JNI_NATIVE_TESTS "Build standalone JNI allocation-path tests" OFF)
if(WEBP_JNI_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Sanitizer validation uses Clang or GCC on Linux/macOS")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WEBP_BUILD_ANIM_UTILS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_CWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_DWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_GIF2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_IMG2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_VWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPINFO OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPMUX OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libwebp GIT_REPOSITORY https://github.com/webmproject/libwebp.git GIT_TAG v1.6.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(libwebp)
add_library(webpjni SHARED native/webp_jni.c)
target_include_directories(webpjni PRIVATE ${JNI_INCLUDE_DIRS} ${libwebp_SOURCE_DIR}/src)
target_link_libraries(webpjni PRIVATE webp)
target_compile_features(webpjni PRIVATE c_std_99)
if(MSVC)
  target_compile_options(webpjni PRIVATE /W4)
else()
  target_compile_options(webpjni PRIVATE -Wall -Wextra -Werror)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(PLATFORM_OS macos)
elseif(WIN32)
  set(PLATFORM_OS windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(PLATFORM_OS linux)
else()
  message(FATAL_ERROR "Unsupported operating system")
endif()
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" PROCESSOR)
if(PROCESSOR MATCHES "^(arm64|aarch64)$")
  set(PLATFORM_ARCH aarch64)
elseif(PROCESSOR MATCHES "^(amd64|x86_64)$")
  set(PLATFORM_ARCH x86_64)
else()
  message(FATAL_ERROR "Unsupported architecture: ${PROCESSOR}")
endif()
install(TARGETS webpjni LIBRARY DESTINATION native/${PLATFORM_OS}-${PLATFORM_ARCH} COMPONENT webpjni RUNTIME DESTINATION native/${PLATFORM_OS}-${PLATFORM_ARCH} COMPONENT webpjni)
install(FILES ${libwebp_SOURCE_DIR}/COPYING ${libwebp_SOURCE_DIR}/PATENTS ${libwebp_SOURCE_DIR}/AUTHORS DESTINATION META-INF/licenses/libwebp COMPONENT webpjni)

if(WEBP_JNI_NATIVE_TESTS)
  enable_testing()
  add_executable(jni_memory_test native/tests/jni_memory_test.c)
  target_include_directories(jni_memory_test PRIVATE ${JNI_INCLUDE_DIRS} ${libwebp_SOURCE_DIR}/src)
  target_link_libraries(jni_memory_test PRIVATE webp)
  target_compile_features(jni_memory_test PRIVATE c_std_99)
  # Test assertions must stay active in Release builds too.
  if(MSVC)
    target_compile_options(jni_memory_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(jni_memory_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  endif()
  add_test(NAME jni_memory_paths COMMAND jni_memory_test)
  set_tests_properties(jni_memory_paths PROPERTIES TIMEOUT 180)
endif()
